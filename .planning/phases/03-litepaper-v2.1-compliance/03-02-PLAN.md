     ```javascript
     // Matchmaking queues — keyed by "modeKey:format" (e.g., "quick_match:1")
     // Each entry: { socketId, wallet, name, color, wager, format, matchMode, queuedAt }
     const matchmakingQueues = new Map();

     function getQueueKey(matchMode, matchLength) {
         return `${matchMode}:${matchLength}`;
     }

     function removeFromAllQueues(socketId) {
         for (const [key, queue] of matchmakingQueues.entries()) {
             const idx = queue.findIndex(e => e.socketId === socketId);
             if (idx !== -1) {
                 queue.splice(idx, 1);
                 if (queue.length === 0) matchmakingQueues.delete(key);
             }
         }
     }
     ```

  2. Add `joinQueue` socket handler inside the `io.on('connection', (client) => { ... })` block:
     ```javascript
     client.on('joinQueue', async (data) => {
         const { matchMode, matchLength, wager, playerName, tankColor } = data;

         // Validate mode (custom_challenge cannot queue — must use createRoom)
         if (matchMode === 'custom_challenge') {
             client.emit('queueError', { reason: 'Custom Challenge uses room codes, not queue' });
             return;
         }

         // Validate using validateMatchMode from solana.js
         const validation = validateMatchMode(matchMode, wager, matchLength);
         if (!validation.valid) {
             client.emit('queueError', { reason: validation.reason });
             return;
         }

         // Remove from any existing queue first
         removeFromAllQueues(client.id);

         const queueKey = getQueueKey(matchMode, matchLength);
         if (!matchmakingQueues.has(queueKey)) {
             matchmakingQueues.set(queueKey, []);
         }
         const queue = matchmakingQueues.get(queueKey);

         // Check if there's already someone waiting
         if (queue.length > 0) {
             const opponent = queue.shift();
             if (queue.length === 0) matchmakingQueues.delete(queueKey);

             // Auto-create room — replicate the FULL createRoom+joinRoom init pattern
             // IMPORTANT: Must match the exact room shape used by createRoom handler (line ~842)
             const roomId = crypto.randomBytes(4).toString('hex');
             const roundType = matchLength === 5 ? 'BO5' : matchLength === 3 ? 'BO3' : '1';
             const roomData = {
                 roomId: roomId,
                 host: {
                     name: opponent.name,
                     color: opponent.color,
                     socketId: opponent.socketId,
                     isReady: false,
                     playAgain: false,
                 },
                 active: false,
                 wager: wager,
                 matchMode: matchMode,
                 totalRounds: matchLength,
             };

             // Initialize wager state (same as createRoom handler line ~844)
             wagerStates[roomId] = {
                 amount: wager,
                 wallets: { [opponent.socketId]: opponent.wallet },
             };

             // Initialize match state (same as createRoom handler line ~851)
             matchStates[roomId] = createMatchState(roomId, roundType);

             // Add to room cache (same as createRoom handler line ~876)
             rooms.set(roomId, roomData);

             // Join both players to the Socket.IO room
             const opponentSocket = io.sockets.sockets.get(opponent.socketId);
             if (opponentSocket) {
                 opponentSocket.roomId = roomId;
                 opponentSocket.join(roomId);
             }
             client.roomId = roomId;
             client.join(roomId);

             // Update room with second player (same shape as joinRoom handler)
             roomData.player = {
                 name: playerName,
                 color: tankColor,
                 socketId: client.id,
                 isReady: false,
                 playAgain: false,
             };
             roomData.active = true;

             // Track second player's wager wallet
             const joinerWallet = authenticatedWallets[client.id] || null;
             wagerStates[roomId].wallets[client.id] = joinerWallet;

             // Trigger escrow creation if wager > 0 (same as joinRoom flow)
             if (wager > 0 && isEscrowEnabled()) {
                 try {
                     await createMatchEscrow(roomId, wager);
                     // Build deposit TXs for both players (same as joinRoom)
                     if (opponent.wallet) {
                         const hostDepositTx = await buildDepositTransaction(opponent.wallet, roomId, wager);
                         if (opponentSocket) opponentSocket.emit('escrowDeposit', { tx: hostDepositTx, wager });
                     }
                     if (joinerWallet) {
                         const playerDepositTx = await buildDepositTransaction(joinerWallet, roomId, wager);
                         client.emit('escrowDeposit', { tx: playerDepositTx, wager });
                     }
                 } catch (err) {
                     console.error(`[Queue] Escrow creation failed for ${roomId}:`, err.message);
                     // Continue without escrow — settlement will use direct transfer fallback
                 }
             }

             broadcastRooms(io);
             trackMatchCreated();
             if (wager > 0) trackWager(wager * 2);

             // Emit queueMatched to both players with room info
             const matchData = {
                 roomId,
                 matchMode,
                 matchLength,
                 wager,
                 host: { name: opponent.name, color: opponent.color },
                 player: { name: playerName, color: tankColor },
             };
             if (opponentSocket) opponentSocket.emit('queueMatched', matchData);
             client.emit('queueMatched', matchData);

             console.log(`[Queue] Matched: ${opponent.name} vs ${playerName} in ${matchMode} BO${matchLength} (${wager} SOL)`);
         } else {
             // No match — add to queue and wait
             queue.push({
                 socketId: client.id,
                 wallet: authenticatedWallets[client.id] || null,
                 name: playerName,
                 color: tankColor,
                 wager: wager,
                 format: matchLength,
                 matchMode: matchMode,
                 queuedAt: Date.now(),
             });
             client.emit('queueWaiting', { matchMode, matchLength, position: queue.length });
             console.log(`[Queue] ${playerName} queued for ${matchMode} BO${matchLength} (${wager} SOL) — ${queue.length} in queue`);
         }
     });
     ```
