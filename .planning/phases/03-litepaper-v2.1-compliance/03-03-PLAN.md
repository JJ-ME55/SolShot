  3. Rewrite `recordMatchPlayed` to accept enriched context and use new milestones:
     ```javascript
     /**
      * Record a completed match and check for v2.1 SHOT milestones.
      *
      * @param {string} walletAddress
      * @param {object} matchContext - Enriched context from main.js settlement:
      *   {
      *     turnCount: number,          // farming protection — min 4 turns
      *     matchId: string,            // dedup
      *     isWagered: boolean,         // true if wager > 0
      *     isWinner: boolean,          // true if this player won the match
      *     maxRoundDamage: number,     // highest damage dealt in any single round
      *     weaponsUsed: number[],      // weapon IDs used by this player
      *   }
      * @returns {{ earned: number, milestones: string[], newBalance: number }}
      */
     export function recordMatchPlayed(walletAddress, matchContext = {}) {
         const state = getPlayerShotState(walletAddress);
         if (!state) return { earned: 0, milestones: [], newBalance: 0 };

         const {
             turnCount = 0,
             matchId = null,
             isWagered = false,
             isWinner = false,
             maxRoundDamage = 0,
             weaponsUsed = [],
         } = matchContext;

         // H033: Farming protection — minimum turns
         if (turnCount < 4) {
             return { earned: 0, milestones: [], newBalance: state.balance, reason: 'Match too short for rewards' };
         }

         // H033: Farming protection — 30-second cooldown
         const now = Date.now();
         if (state.lastRewardAt && (now - state.lastRewardAt) < 30_000) {
             return { earned: 0, milestones: [], newBalance: state.balance, reason: 'Reward cooldown active' };
         }

         // H033: Farming protection — match ID dedup
         if (matchId && state.claimedMatchIds && state.claimedMatchIds.has(matchId)) {
             return { earned: 0, milestones: [], newBalance: state.balance, reason: 'Match already claimed' };
         }

         // H034: Check global supply cap
         if (totalShotEmitted >= SHOT_TOKEN_CONFIG.rewardPool) {
             return { earned: 0, milestones: [], newBalance: state.balance, reason: 'Reward pool exhausted' };
         }

         // Update player stats
         state.totalMatchesPlayed++;
         state.matchesPlayed++; // legacy compat
         if (isWagered) {
             state.wageredMatchesPlayed++;
             if (isWinner) state.wageredWins++;
         }
         if (isWinner) {
             state.consecutiveWins++;
         } else {
             state.consecutiveWins = 0; // reset streak on loss
         }

         // Check for prestige weapon usage
         const usedNoPrestige = weaponsUsed.length > 0 &&
             !weaponsUsed.some(id => PRESTIGE_WEAPON_IDS.includes(id));

         // Build context for milestone checks
         const ctx = { isWinner, maxRoundDamage, usedNoPrestige, isWagered };

         // Check all milestones
         let totalEarned = 0;
         const earnedMilestones = [];

         for (const milestone of SHOT_MILESTONES) {
             // Skip already earned
             if (state.milestonesEarned.includes(milestone.id)) continue;
             // Run check
             if (milestone.check(state, ctx)) {
                 let reward = milestone.reward;
                 // LP-05: Practice mode emits at 25% rate
                 if (!isWagered) {
                     reward = Math.floor(reward * 0.25);
                     if (reward < 1) reward = 1; // minimum 1 SHOT
                 }
                 state.milestonesEarned.push(milestone.id);
                 state.balance += reward;
                 totalEarned += reward;
                 earnedMilestones.push(milestone.label);
             }
         }

         // H034: Clamp earned to remaining supply
         if (totalShotEmitted + totalEarned > SHOT_TOKEN_CONFIG.rewardPool) {
             const allowed = SHOT_TOKEN_CONFIG.rewardPool - totalShotEmitted;
             const excess = totalEarned - allowed;
             state.balance -= excess;
             totalEarned = allowed;
         }

         // H034: Track global emissions
         totalShotEmitted += totalEarned;
         if (totalEarned > 0) persistEmissionCount();

         // H033: Update anti-farming state
         state.lastRewardAt = now;
         if (!state.claimedMatchIds) state.claimedMatchIds = new Set();
         if (matchId) state.claimedMatchIds.add(matchId);

         return {
             earned: totalEarned,
             milestones: earnedMilestones,
             newBalance: state.balance,
             totalMatchesPlayed: state.totalMatchesPlayed,
         };
     }
     ```
