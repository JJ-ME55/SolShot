const { message, signature, timestamp } = data
if (!message || !signature || !timestamp) {
    client.emit('rejoinError', { reason: 'Signature required for rejoin' })
    return

// Verify the auth message format and timestamp
const msgCheck = verifyAuthMessage(message, walletAddress, timestamp)
if (!msgCheck.valid) {
    client.emit('rejoinError', { reason: msgCheck.reason || 'Invalid auth message' })
    return

// Verify the Ed25519 signature

  <action>
In `client/src/App.js`, the `attemptRejoin` function (line ~52) currently does:
```js
const attemptRejoin = () => {
    const walletAddress = window.solWallet?.publicKey?.toString();
    if (walletAddress) {
        window.socket.emit('rejoinRoom', { walletAddress });
    }
};
```

Replace with an async version that signs a message before emitting:
```js
const attemptRejoin = async () => {
    const walletAddress = window.solWallet?.publicKey?.toString();
    const signMessage = window.solWallet?.signMessage;
    if (!walletAddress || !signMessage) {
        // Wallet not ready yet — skip rejoin attempt (30s window allows retry on reconnect)
        return;
    }

    try {
        const timestamp = Date.now();
        const message = `SolShot Auth: ${walletAddress} at ${timestamp}`;
        const encodedMessage = new TextEncoder().encode(message);
        const signature = await signMessage(encodedMessage);
        const signatureBase64 = btoa(String.fromCharCode(...signature));

        window.socket.emit('rejoinRoom', {
            walletAddress,
            message,
            signature: signatureBase64,
            timestamp,
        });
    } catch (err) {
        console.warn('[SolShot] Rejoin signature failed:', err.message);
        // Don't block — if signMessage fails (user rejects), the 30s reconnect window
        // allows another attempt on next socket connect event
    }
};
```

This mirrors the exact signing pattern used in `WalletContext.authenticate()`:
- Same message format: `SolShot Auth: ${walletAddress} at ${timestamp}`
- Same signature encoding: base64 via `btoa(String.fromCharCode(...))`
- Same timestamp included for server-side freshness validation

The `signMessage` function is accessed from `window.solWallet` which is set by WalletContext. If the wallet is not yet initialized (common on page load), `attemptRejoin` returns early without emitting. The socket `connect` event will retry when the socket reconnects, and by then the wallet adapter should be ready.

Note: The `window.socket.on('connect', attemptRejoin)` line is fine with async callbacks — Socket.IO ignores the return value of event listeners.
